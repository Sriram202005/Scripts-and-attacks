NMAP - ADVANCED SCRIPTS
Initial Recon
bash# Full port scan first (fast)
nmap -p- --min-rate=10000 -T4 target -oN allports.txt

# Targeted version + script scan on open ports
nmap -sC -sV -p 21,22,80,139,443,445,3389 target -oN targeted.txt

# UDP scan (often missed in CTF)
nmap -sU --top-ports 200 --min-rate=5000 target

# OS detection + aggressive
nmap -A -T4 target

# Vuln scan all open ports
nmap --script vuln -p 21,22,80,445 target

LDAP
Nmap Scripts
bashnmap --script ldap-rootdse,ldap-search,ldap-novell-getpass -p 389,636 target

Tools
bash# Anonymous LDAP enum
ldapsearch -x -H ldap://target -b "dc=domain,dc=com"
ldapsearch -x -H ldap://target -b "" -s base namingcontexts  # get base DN

# Authenticated
ldapsearch -x -H ldap://target -D "cn=admin,dc=domain,dc=com" -w pass -b "dc=domain,dc=com"

# Dump everything
ldapsearch -x -H ldap://target -D "user@domain.com" -w pass -b "dc=domain,dc=com" > ldap_dump.txt

# Specific queries
ldapsearch -x -H ldap://target -b "dc=domain,dc=com" "(objectClass=user)" sAMAccountName  # all users
ldapsearch -x -H ldap://target -b "dc=domain,dc=com" "(objectClass=group)"               # groups
ldapsearch -x -H ldap://target -b "dc=domain,dc=com" "(objectClass=computer)"            # computers
ldapsearch -x -H ldap://target -b "dc=domain,dc=com" "(&(objectClass=user)(adminCount=1))" # admins

# ldapdomaindump (outputs HTML/JSON)
ldapdomaindump ldap://target -u 'domain\user' -p pass

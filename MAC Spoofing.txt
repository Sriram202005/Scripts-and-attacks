# 1. Check current MAC
ip link show wlan0

# 2. Bring interface down
sudo ip link set dev wlan0 down

# 3. Change MAC (example: 00:11:22:33:44:55)
sudo ip link set dev wlan0 address 00:11:22:33:44:55

# 4. Bring interface up again
sudo ip link set dev wlan0 up

# 5. Verify new MAC
ip link show wlan0

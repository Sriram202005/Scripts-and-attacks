PORT 389/636 — LDAP/LDAPS
bashnmap --script ldap-rootdse,ldap-search -p 389 target

# Anonymous
ldapsearch -x -H ldap://DC_IP -b "dc=domain,dc=com"
ldapsearch -x -H ldap://DC_IP -b "" -s base namingcontexts

# Authenticated queries
ldapsearch -x -H ldap://DC_IP -D "user@domain.com" -w pass -b "dc=domain,dc=com" "(objectClass=user)" sAMAccountName
ldapsearch -x -H ldap://DC_IP -D "user@domain.com" -w pass -b "dc=domain,dc=com" "(&(objectClass=user)(adminCount=1))"  # admins
ldapsearch -x -H ldap://DC_IP -D "user@domain.com" -w pass -b "dc=domain,dc=com" "(objectClass=computer)"
ldapsearch -x -H ldap://DC_IP -D "user@domain.com" -w pass -b "dc=domain,dc=com" "(ms-MCS-AdmPwd=*)"  # LAPS passwords
